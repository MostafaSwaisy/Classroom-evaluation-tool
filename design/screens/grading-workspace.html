<?php namespace App\Http\Controllers; use Illuminate\Http\Request; use App\Models\User; use Firebase\JWT\JWT; use Firebase\JWT\Key; use Illuminate\Support\Facades\Hash; class AuthController extends Controller { // دالة تسجيل الدخول وإنشاء الـ JWT Claims للطالب public function login(Request $request) { $credentials = $request->validate([ 'email' => 'required|email', 'password' => 'required|string|min:6', ]); $user = User::where('email', $credentials['email'])->first(); if (!$user || !Hash::check($credentials['password'], $user->password)) { return response()->json([ 'status' => 'error', 'message' => 'بيانات الدخول غير صحيحة', ], 401); } // بناء كائن الـ Payload بما يتوافق مع معيار RFC 7519 $issuedAt = time(); $expire = $issuedAt + (60 * 60); // صالح لمدة ساعة واحدة $payload = [ 'iss' => config('app.url'), 'iat' => $issuedAt, 'exp' => $expire, 'sub' => $user->id, 'role' => $user->role ?? 'student', ]; $token = JWT::encode($payload, config('jwt.secret'), 'HS256'); return response()->json([ 'status' => 'success', 'token' => $token, 'type' => 'Bearer', ]); } }
